All endpoints require auth via session cookie or a Bearer token from /api/token.

Authentication
POST/api/token
Body: {"username":"...","password":"..."} → bearer token.
Runs
POST/api/runs
Create a run. Returns run_id.
{"name":"cifar100-exp-01","dataset":"cifar100","methods":["DLG","iDLG"], "num_dummy":1,"iterations":300,"lr":1.0,"num_classes":100}
GET/api/runs
List all runs.
GET/api/runs/:run_id
Get metadata for a single run.
Upload
POST/api/runs/:run_id/metrics
Upload metric series or stream per‑iteration values. The recommended payload uses series, append and scalars keys. Legacy keys (loss, mse, append_loss, append_mse) are still accepted for backwards compatibility.
# Full series at end of run {"method":"iDLG","series":{"loss":[0.5,0.3,...],"mse":[0.2,0.1,...]}, "scalars":{"final_loss":0.000042,"final_mse":0.00012,"label":7,"gt_label":7}} # Stream per iteration {"method":"DLG","append":{"loss":0.3421,"mse":0.1204}}
POST/api/runs/:run_id/images
Multipart upload (file field) or base64 JSON. Accepts .png, .jpg, .jpeg, .gif and .webp files.
POST/api/runs/:run_id/data
Multipart file upload. Accepts .pkl, .npy, .npz, .json, .csv, .txt and .log files.
POST/api/runs/:run_id/logs
Append to a log file. Set "append": false to overwrite.
{"filename":"run.log","content":"iter 50 loss=0.34","append":true}
Python Client
import requests, os SERVER = "http://localhost:5000" s = requests.Session() tok = s.post(f"{SERVER}/api/token", json={"username":"admin","password":"changeme"}).json()["token"] s.headers["Authorization"] = f"Bearer {tok}" def create_run(name, dataset, methods, **kw): return s.post(f"{SERVER}/api/runs", json={"name": name, "dataset": dataset, "methods": methods, **kw}).json()["run_id"] def upload_metrics_full(run_id, method, loss, mse, **scalars): # Upload full arrays and optional scalar metadata s.post(f"{SERVER}/api/runs/{run_id}/metrics", json={"method": method, "series": {"loss": loss, "mse": mse}, "scalars": scalars}) def upload_metric_point(run_id, method, loss, mse): # Stream a single metric point s.post(f"{SERVER}/api/runs/{run_id}/metrics", json={"method": method, "append": {"loss": loss, "mse": mse}}) def upload_image(run_id, path): with open(path, "rb") as f: s.post(f"{SERVER}/api/runs/{run_id}/images", files={"file": (os.path.basename(path), f, "image/png")}) def upload_data(run_id, path): with open(path, "rb") as f: s.post(f"{SERVER}/api/runs/{run_id}/data", files={"file": (os.path.basename(path), f)}) def log(run_id, msg, filename="run.log"): s.post(f"{SERVER}/api/runs/{run_id}/logs", json={"filename": filename, "content": msg, "append": True}) # Example usage: # run_id = create_run("cifar100-run-1", "cifar100", ["DLG", "iDLG"], num_dummy=1, iterations=300, lr=1.0, num_classes=100) # upload_metric_point(run_id, "DLG", loss=0.3421, mse=0.1204) # log(run_id, f"iteration 1 loss={0.3421}") # upload_image(run_id, "path/to/plot.png") # upload_metrics_full(run_id, "DLG", loss_array, mse_array, final_loss=loss_array[-1], final_mse=mse_array[-1], label=label, gt_label=int(gt_label))
